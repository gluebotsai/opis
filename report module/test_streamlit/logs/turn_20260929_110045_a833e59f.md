MATCH (c:Conversation)
WHERE c.created_at IS NOT NULL AND c.ended_at IS NOT NULL
RETURN coalesce(c.channel, 'all') AS channel,
       count(c) AS conversations_measured,
       round(avg(duration.between(c.created_at, c.ended_at).seconds), 1) AS avg_handle_time_seconds,
       round(avg(duration.between(c.created_at, c.ended_at).seconds) / 60.0, 2) AS avg_handle_time_minutes,
       min(duration.between(c.created_at, c.ended_at).seconds) AS min_duration_seconds,

ORDER BY conversations_measured DESC
LIMIT 50
