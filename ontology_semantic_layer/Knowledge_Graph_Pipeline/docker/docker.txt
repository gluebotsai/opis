# ==============================================================================
# CDC STREAMING STACK - DOCKER RUN COMMANDS (STANDALONE)
# ==============================================================================

# 1. CREATE SHARED DOCKER NETWORK

docker network create cdc-network

# 2. RUN KAFKA (PORT 9094, RESTART ON FAILURE)

docker run -d \
  --name cdc-kafka \
  --network cdc-network \
  --restart on-failure \
  -p 9094:9094 \
  -e KAFKA_NODE_ID=1 \
  -e KAFKA_PROCESS_ROLES=broker,controller \
  -e KAFKA_CONTROLLER_QUORUM_VOTERS=1@cdc-kafka:29093 \
  -e KAFKA_LISTENERS=PLAINTEXT://0.0.0.0:29092,CONTROLLER://0.0.0.0:29093,EXTERNAL://0.0.0.0:9094 \
  -e KAFKA_ADVERTISED_LISTENERS=PLAINTEXT://cdc-kafka:29092,EXTERNAL://0.0.0.0:9094 \
  -e KAFKA_LISTENER_SECURITY_PROTOCOL_MAP=CONTROLLER:PLAINTEXT,PLAINTEXT:PLAINTEXT,EXTERNAL:PLAINTEXT \
  -e KAFKA_INTER_BROKER_LISTENER_NAME=PLAINTEXT \
  -e KAFKA_CONTROLLER_LISTENER_NAMES=CONTROLLER \
  -e KAFKA_OFFSETS_TOPIC_REPLICATION_FACTOR=1 \
  -e KAFKA_TRANSACTION_STATE_LOG_REPLICATION_FACTOR=1 \
  -e KAFKA_TRANSACTION_STATE_LOG_MIN_ISR=1 \
  -e KAFKA_GROUP_INITIAL_REBALANCE_DELAY_MS=0 \
  -e KAFKA_AUTO_CREATE_TOPICS_ENABLE="true" \
  -e CLUSTER_ID=MkU3OEVBNTcwNTJENDM2Qk \
  -e KAFKA_MESSAGE_MAX_BYTES=10485760 \
  -e KAFKA_REPLICA_FETCH_MAX_BYTES=10485760 \
  confluentinc/cp-kafka:7.6.1

# 3. RUN DEBEZIUM / KAFKA CONNECT (PORT 8083, RESTART ON FAILURE)

docker run -d \
  --name cdc-connect \
  --network cdc-network \
  --restart on-failure \
  -p 8083:8083 \
  -e BOOTSTRAP_SERVERS=cdc-kafka:29092 \
  -e GROUP_ID=cdc-connect \
  -e CONFIG_STORAGE_TOPIC=_connect_configs \
  -e OFFSET_STORAGE_TOPIC=_connect_offsets \
  -e STATUS_STORAGE_TOPIC=_connect_status \
  -e CONFIG_STORAGE_REPLICATION_FACTOR=1 \
  -e OFFSET_STORAGE_REPLICATION_FACTOR=1 \
  -e STATUS_STORAGE_REPLICATION_FACTOR=1 \
  -e KEY_CONVERTER=org.apache.kafka.connect.json.JsonConverter \
  -e VALUE_CONVERTER=org.apache.kafka.connect.json.JsonConverter \
  -e CONNECT_KEY_CONVERTER_SCHEMAS_ENABLE="true" \
  -e CONNECT_VALUE_CONVERTER_SCHEMAS_ENABLE="true" \
  -e CONNECT_PRODUCER_MAX_REQUEST_SIZE=10485760 \
  -e CONNECT_CONSUMER_MAX_PARTITION_FETCH_BYTES=10485760 \
  debezium/connect:2.7.3.Final
